# ——— Homeboard: da aggiungere in fondo a /boot/firmware/config.txt (lo fa install.sh) ———

# I due tasti del televisore: ognuno fra un GPIO e GND, pull-up interno, attivo basso.
# Arrivano come tasti di una tastiera (keycode Linux: KEY_1 = 2, KEY_2 = 3); li legge voice.
# GPIO (numerazione BCM) scelti lontani da I2C, SPI e UART; pin fisici fra parentesi.
dtoverlay=gpio-key,gpio=5,active_low=1,gpio_pull=up,keycode=2,label="microfono"  # pin 29
dtoverlay=gpio-key,gpio=6,active_low=1,gpio_pull=up,keycode=3,label="parla"      # pin 31

# Watchdog hardware: se il sistema si pianta, il Pi si riavvia da solo (con RuntimeWatchdogSec in systemd).
dtparam=watchdog=on

# Niente schermata arcobaleno all'avvio.
disable_splash=1
# ——— fine Homeboard ———
